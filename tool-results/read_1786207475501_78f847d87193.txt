    63→    63→function loadSession(): Session | null {
    64→    64→  if (typeof window === "undefined") return null;
    65→    65→  try {
    66→    66→    const raw = sessionStorage.getItem(STORAGE_KEY);
    67→    67→    if (!raw) return null;
    68→    68→    return JSON.parse(raw);
    69→    69→  } catch {
    70→    70→    return null;
    71→    71→  }
    72→    72→}
    73→    73→function saveSession(s: Session) {
    74→    74→  if (typeof window !== "undefined")
    75→    75→    sessionStorage.setItem(STORAGE_KEY, JSON.stringify(s));
    76→    76→}
    77→    77→function clearSession() {
    78→    78→  if (typeof window !== "undefined") sessionStorage.removeItem(STORAGE_KEY);
    79→    79→}
    80→    80→function getRoomCodeFromURL(): string | null {
    81→    81→  if (typeof window === "undefined") return null;

    83→    83→  return params.get("room") || null;
    84→    84→}
    85→    85→function formatTime(s: number): string {
    86→    86→  const m = Math.floor(s / 60);
    87→    87→  const sec = s % 60;
    88→    88→  return `${m}:${sec.toString().padStart(2, "0")}`;
    89→    89→}
