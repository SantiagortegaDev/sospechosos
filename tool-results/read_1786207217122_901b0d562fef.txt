    61→function loadSession(): Session | null {
    62→  if (typeof window === "undefined") return null;
    63→  try {
    64→    const raw = sessionStorage.getItem(STORAGE_KEY);
    65→    if (!raw) return null;
    66→    return JSON.parse(raw);
    67→  } catch {
    68→    return null;
    69→  }
    70→}
    71→function saveSession(s: Session) {
    72→  if (typeof window !== "undefined")
    73→    sessionStorage.setItem(STORAGE_KEY, JSON.stringify(s));
    74→}
    75→function clearSession() {
    76→  if (typeof window !== "undefined") sessionStorage.removeItem(STORAGE_KEY);
    77→}
    78→function getRoomCodeFromURL(): string | null {
    79→  if (typeof window === "undefined") return null;
    80→  const params = new URLSearchParams(window.location.search);
    81→  return params.get("room") || null;
    82→}
    83→function formatTime(s: number): string {
    84→  const m = Math.floor(s / 60);
    85→  const sec = s % 60;
    86→  return `${m}:${sec.toString().padStart(2, "0")}`;
    87→}
