    61→    61→function loadSession(): Session | null {
    62→    62→  if (typeof window === "undefined") return null;
    63→    63→  try {
    64→    64→    const raw = sessionStorage.getItem(STORAGE_KEY);
    65→    65→    if (!raw) return null;
    66→    66→    return JSON.parse(raw);
    67→    67→  } catch {
    68→    68→    return null;
    69→    69→  }
    70→    70→}
    71→    71→function saveSession(s: Session) {
    72→    72→  if (typeof window !== "undefined")
    73→    73→    sessionStorage.setItem(STORAGE_KEY, JSON.stringify(s));
    74→    74→}
    75→    75→function clearSession() {
    76→    76→  if (typeof window !== "undefined") sessionStorage.removeItem(STORAGE_KEY);
    77→    77→}
