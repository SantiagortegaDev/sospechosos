    63→function loadSession(): Session | null {
    64→  if (typeof window === "undefined") return null;
    65→  try {
    66→    const raw = sessionStorage.getItem(STORAGE_KEY);
    67→    if (!raw) return null;
    68→    return JSON.parse(raw);
    69→  } catch {
    70→    return null;
    71→  }
    72→}
    73→function saveSession(s: Session) {
    74→  if (typeof window !== "undefined")
    75→    sessionStorage.setItem(STORAGE_KEY, JSON.stringify(s));
    76→}
    77→function clearSession() {
    78→  if (typeof window !== "undefined") sessionStorage.removeItem(STORAGE_KEY);
    79→}
    80→function getRoomCodeFromURL(): string | null {
    81→  if (typeof window === "undefined") return null;
    82→  const params = new URLSearchParams(window.location.search);
    83→  return params.get("room") || null;
    84→}
    85→function formatTime(s: number): string {
    86→  const m = Math.floor(s / 60);
    87→  const sec = s % 60;
    88→  return `${m}:${sec.toString().padStart(2, "0")}`;
    89→}
